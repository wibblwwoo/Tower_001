using Godot;
using System;
using System.Collections.Generic;

/// <summary>
/// Manages a process that counts from 1 to 100 with configurable heartbeat increments
/// and difficulty scaling.
/// </summary>
public partial class ProcessHeartbeatSystem : Node
{
    // Signals (Godot events)
    [Signal]
    public delegate void ProcessCompletedEventHandler();
    
    [Signal]
    public delegate void ProcessValueChangedEventHandler(float displayValue, float actualValue);

    // Constants
    private const float BASE_MAX_VALUE = 100f;
    private const float HEARTBEAT_INTERVAL = 0.5f; // seconds
    
    // Configuration properties
    public float DifficultyScale { get; set; } = 0f;
    public float CurrentIncrementValue { get; private set; } = 1f;
    public float IncrementGrowthRate { get; set; } = 0f; // How much the increment grows per beat
    
    // State tracking
    private float _currentValue = 1f;
    private float _actualMaxValue;
    private float _timer = 0f;
    private bool _isActive = false;
    
    // Called when the node enters the scene tree
    public override void _Ready()
    {
        ResetProcess();
    }
    
    // Called every frame
    public override void _Process(double delta)
    {
        if (!_isActive) return;
        
        _timer += (float)delta;
        
        // Check if it's time for a heartbeat
        if (_timer >= HEARTBEAT_INTERVAL)
        {
            ProcessHeartbeat();
            _timer = 0f;
        }
    }
    
    /// <summary>
    /// Processes a single heartbeat, updating values and checking completion
    /// </summary>
    private void ProcessHeartbeat()
    {
        _currentValue += CurrentIncrementValue;
        CurrentIncrementValue += IncrementGrowthRate;
        
        // Calculate display value (scaled back to 1-100 range)
        float displayValue = (_currentValue / _actualMaxValue) * BASE_MAX_VALUE;
        
        // Emit signal with both display and actual values
        EmitSignal(SignalName.ProcessValueChanged, displayValue, _currentValue);
        
        // Check if process is complete
        if (_currentValue >= _actualMaxValue)
        {
            EmitSignal(SignalName.ProcessCompleted);
            ResetProcess();
        }
    }
    
    /// <summary>
    /// Resets the process to initial state
    /// </summary>
    public void ResetProcess()
    {
        _currentValue = 1f;
        CurrentIncrementValue = 1f;
        _actualMaxValue = BASE_MAX_VALUE * (1f + DifficultyScale);
        _timer = 0f;
    }
    
    /// <summary>
    /// Starts the process
    /// </summary>
    public void StartProcess()
    {
        _isActive = true;
    }
    
    /// <summary>
    /// Stops the process
    /// </summary>
    public void StopProcess()
    {
        _isActive = false;
    }
    
    /// <summary>
    /// Calculates the estimated time to completion based on current settings
    /// </summary>
    /// <returns>Estimated time in seconds</returns>
    public float CalculateEstimatedCompletionTime()
    {
        float remainingValue = _actualMaxValue - _currentValue;
        float beats = 0;
        float currentIncrement = CurrentIncrementValue;
        float accumulator = 0f;
        
        while (accumulator < remainingValue)
        {
            accumulator += currentIncrement;
            currentIncrement += IncrementGrowthRate;
            beats++;
        }
        
        return beats * HEARTBEAT_INTERVAL;
    }
}
