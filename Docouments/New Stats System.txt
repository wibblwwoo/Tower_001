```csharp
public enum UnlockType

    Feature,
    Content,

public abstract class BaseUnlock
{
    public string Id { get; }
    public string Name { get; }
    public UnlockType Type { get; }
    public UnlockRequirement[] Requirements { get; }
    public UnlockReward Reward { get; }
    
    protected BaseUnlock(
        string name, 
        UnlockType type,
        UnlockRequirement[] requirements,
        UnlockReward reward)
    {
        Id = Guid.NewGuid().ToString();
        Name = name;
        Type = type;
        Requirements = requirements;
        Reward = reward;
    }

    public virtual bool CheckRequirements(ProgressionData data)
    {
        return Requirements.All(r => r.IsMet(data));
    }

    public virtual void Apply(string characterId)
    {
        Reward?.Apply(characterId);
    }
}

public class FeatureUnlock : BaseUnlock
{
    public string FeatureId { get; }
    
    public FeatureUnlock(
        string name,
        string featureId,
        UnlockRequirement[] requirements,
        UnlockReward reward) 
        : base(name, UnlockType.Feature, requirements, reward)
    {
        FeatureId = featureId;
    }
}

public class StatUnlock : BaseUnlock
{
    public StatType StatType { get; }
    public float BonusValue { get; }
    
    public StatUnlock(
        string name,
        StatType statType,
        float bonusValue,
        UnlockRequirement[] requirements,
        UnlockReward reward) 
        : base(name, UnlockType.Stat, requirements, reward)
    {
        StatType = statType;
        BonusValue = bonusValue;
    }

    {
        base.Apply(characterId);

}
```

### B. Milestone System

```csharp
public enum MilestoneType
{
    Progression,
    Achievement,
    Prestige,
    Ascension,
    Feature,
    Stat
}

public abstract class BaseMilestone
{
    public string Id { get; }
    public string Name { get; }
    public MilestoneType Type { get; }
    public bool Repeatable { get; }
    public int RequiredLevel { get; }
    public MilestoneReward Reward { get; }
    public bool IsCompleted { get; private set; }
    public int CompletionCount { get; private set; }

    protected BaseMilestone(
        string name,
        MilestoneType type,
        int requiredLevel,
        MilestoneReward reward,
        bool repeatable = false)
    {
        Id = Guid.NewGuid().ToString();
        Name = name;
        Type = type;
        RequiredLevel = requiredLevel;
        Reward = reward;
        Repeatable = repeatable;
    }

    public virtual bool CheckRequirements(ProgressionData data)
    {
        if (IsCompleted && !Repeatable)
            return false;

        return CheckSpecificRequirements(data);
    }

    protected abstract bool CheckSpecificRequirements(ProgressionData data);

    public virtual void Complete(string characterId)
    {
        if (IsCompleted && !Repeatable)
            return;

        CompletionCount++;
        IsCompleted = true;
        Reward?.Apply(characterId);
        RaiseMilestoneCompletedEvent(characterId);
    }

    protected virtual void RaiseMilestoneCompletedEvent(string characterId)
    {
        Globals.Instance?.gameMangers?.Events?.RaiseEvent(
            EventType.MilestoneCompleted,
            new MilestoneEventArgs(
                characterId,
                Id,
                Type,
                CompletionCount
            )
        );
    }
}

public class ProgressionMilestone : BaseMilestone
{
    public ProgressionType ProgressionType { get; }

    public ProgressionMilestone(
        string name,
        ProgressionType progressionType,
        int requiredLevel,
        MilestoneReward reward,
        bool repeatable = false)
        : base(name, MilestoneType.Progression, requiredLevel, reward, repeatable)
    {
        ProgressionType = progressionType;
    }

    protected override bool CheckSpecificRequirements(ProgressionData data)
    {
        return ProgressionType switch
        {
            ProgressionType.Prestige => data.PrestigeLevel >= RequiredLevel,
            ProgressionType.Ascension => data.AscensionLevel >= RequiredLevel,
            _ => false
        };
    }
}

public class StatMilestone : BaseMilestone
{
    public StatType StatType { get; }
    public float RequiredValue { get; }

    public StatMilestone(
        string name,
        StatType statType,
        float requiredValue,
        MilestoneReward reward,
        bool repeatable = false)
        : base(name, MilestoneType.Stat, 0, reward, repeatable)
    {
        StatType = statType;
        RequiredValue = requiredValue;
    }

    protected override bool CheckSpecificRequirements(ProgressionData data)
    {
        var character = Globals.Instance.gameMangers.Player.GetCharacter(data.CharacterId);
        float currentValue = character.GetStatValue(StatType);
        return currentValue >= RequiredValue;
    }
}
```

### B. Managers Implementation

```csharp
public class MilestoneManager
{
    private readonly string _characterId;
    private readonly Dictionary<string, BaseMilestone> _milestones = new();
    private readonly HashSet<string> _completedMilestones = new();

    public MilestoneManager(string characterId)
    {
        _characterId = characterId;
    }

    public void RegisterMilestone(BaseMilestone milestone)
    {
        _milestones[milestone.Id] = milestone;
    }

    public void CheckMilestones(ProgressionData data)
    {
        foreach (var milestone in _milestones.Values)
        {
            if (milestone.CheckRequirements(data))
            {
                CompleteMilestone(milestone);
            }
        }
    }

    private void CompleteMilestone(BaseMilestone milestone)
    {
        _completedMilestones.Add(milestone.Id);
        milestone.Complete(_characterId);
    }

    public bool IsMilestoneCompleted(string milestoneId)
    {
        return _completedMilestones.Contains(milestoneId);
    }
}

public class UnlockManager
{
    private readonly string _characterId;
    private readonly Dictionary<string, BaseUnlock> _unlocks = new();
    private readonly HashSet<string> _unlockedFeatures = new();

    public UnlockManager(string characterId)
    {
        _characterId = characterId;
    }

    public void RegisterUnlock(BaseUnlock unlock)
    {
        _unlocks[unlock.Id] = unlock;
    }

    public void CheckUnlocks(ProgressionData data)
    {
        foreach (var unlock in _unlocks.Values)
        {
            if (!_unlockedFeatures.Contains(unlock.Id) && 
                unlock.CheckRequirements(data))
            {
                UnlockFeature(unlock);
            }
        }
    }

    private void UnlockFeature(BaseUnlock unlock)
    {
        _unlockedFeatures.Add(unlock.Id);
        unlock.Apply(_characterId);
    }

    public bool IsUnlocked(string unlockId)
    {
        return _unlockedFeatures.Contains(unlockId);
    }
}
```

```csharp
    [Test]

    {
```

```csharp
        bool testPassed = true;

        try
        {
            var characterId = CreateTestCharacter();
            var character = _playerManager.GetCharacter(characterId);
            var statSystem = character._statSystem;

            // Register unlock requiring 15 base strength
            var unlock = new StatUnlock(
                "Strength Mastery",

            while (character.GetBaseStatValue(StatType.Strength) < targetValue)
```

```csharp
        isValid &= ValidateStats();
        isValid &= ValidateModifiers();
        isValid &= ValidateBonuses();
        isValid &= ValidateMilestones();
        isValid &= ValidateUnlocks();
        return isValid;

    private bool ValidateStats()
    {
        foreach (StatType statType in Enum.GetValues(typeof(StatType)))
        {
            // Validate base values
            float baseValue = _system.GetBaseStatValue(statType);
            if (baseValue < 0)
            {
                _logger.LogError($"Invalid base value for {statType}: {baseValue}");
                return false;
            }

            // Validate calculated values
            float calculatedValue = _system.GetStatValue(statType);
            if (calculatedValue < baseValue)
            {
                _logger.LogError(
                    $"Invalid calculated value for {statType}: " +
                    $"Base={baseValue}, Calculated={calculatedValue}");
                return false;
            }
        }
        return true;
    }

    private bool ValidateModifiers()
    {
        foreach (StatType statType in Enum.GetValues(typeof(StatType)))
        {
            var modifiers = _system.GetModifiers(statType);
            
            // Check for duplicate IDs
            var duplicateIds = modifiers
                .GroupBy(m => m.Id)
                .Where(g => g.Count() > 1)
                .Select(g => g.Key);

            if (duplicateIds.Any())
            {
                _logger.LogError(
                    $"Duplicate modifier IDs found for {statType}: " +
                    string.Join(", ", duplicateIds));
                return false;
            }
        }
        return true;
    }
}
```

### B. Event Throttling System

```csharp
public class EventThrottler
{
    private readonly Dictionary<Type, ThrottleSettings> _settings;
    private readonly Dictionary<Type, Queue<GameEventArgs>> _eventQueues;
    private readonly Dictionary<Type, DateTime> _lastProcessedTime;
    
    public EventThrottler()
    {
        _settings = new Dictionary<Type, ThrottleSettings>();
        _eventQueues = new Dictionary<Type, Queue<GameEventArgs>>();
        _lastProcessedTime = new Dictionary<Type, DateTime>();
        
        InitializeDefaultSettings();
    }

    private void InitializeDefaultSettings()
    {
        // Stat changes - batch every 100ms
        RegisterEventType<StatChangeEventArgs>(
            TimeSpan.FromMilliseconds(100),
            100
        );

        // Milestone progress - batch every 500ms
        RegisterEventType<MilestoneProgressEventArgs>(
            TimeSpan.FromMilliseconds(500),
            50
        );

        // Multiple milestones - process immediately
        RegisterEventType<MultipleMilestoneEventArgs>(
            TimeSpan.Zero,
            1
        );
    }

    public void QueueEvent<T>(T eventArgs) where T : GameEventArgs
    {
        var type = typeof(T);
        if (!_eventQueues.ContainsKey(type))
        {
            _eventQueues[type] = new Queue<GameEventArgs>();
        }

        _eventQueues[type].Enqueue(eventArgs);
        TryProcessQueue(type);
    }

    private void TryProcessQueue(Type eventType)
    {
        if (!_settings.TryGetValue(eventType, out var settings))
            return;

        var now = DateTime.UtcNow;
        if (!_lastProcessedTime.TryGetValue(eventType, out var lastProcessed))
        {
            lastProcessed = DateTime.MinValue;
        }

        if (now - lastProcessed < settings.ThrottleInterval)
            return;

        var queue = _eventQueues[eventType];
        if (queue.Count >= settings.BatchSize || settings.ThrottleInterval == TimeSpan.Zero)
        {
            ProcessEvents(eventType, queue);
            _lastProcessedTime[eventType] = now;
        }
    }

    private void ProcessEvents(Type eventType, Queue<GameEventArgs> queue)
    {
        var events = new List<GameEventArgs>();
        while (queue.Count > 0 && events.Count < _settings[eventType].BatchSize)
        {
            events.Add(queue.Dequeue());
        }

        // Handle based on event type
        if (eventType == typeof(StatChangeEventArgs))
        {
            ProcessStatChangeBatch(events.Cast<StatChangeEventArgs>());
        }
        else if (eventType == typeof(MilestoneProgressEventArgs))
        {
            ProcessMilestoneProgressBatch(events.Cast<MilestoneProgressEventArgs>());
        }
        else
        {
            // Process individual events
            foreach (var evt in events)
            {
                Globals.Instance.gameMangers.Events.RaiseEvent(
                    GetEventTypeForArgs(evt), evt);
            }
        }
    }

    private void ProcessStatChangeBatch(IEnumerable<StatChangeEventArgs> events)
    {
        // Group by character and stat type
        var grouped = events.GroupBy(
            e => (e.CharacterId, e.StatType),
            e => e
        );

        foreach (var group in grouped)
        {
            var first = group.First();
            var last = group.Last();
            
            // Only raise event for net change
            Globals.Instance.gameMangers.Events.RaiseEvent(
                EventType.CharacterStatChanged,
                new StatChangeEventArgs(
                    first.CharacterId,
                    first.StatType,
                    first.OldValue,
                    last.NewValue,
                    ChangeSource.Multiple
                )
            );
        }
    }
}
```

```csharp
            GetStatChangeImportance(args)

        // Don't notify small changes
        float changePercent = Math.Abs(args.NewValue - args.OldValue) / args.OldValue;
        if (changePercent < 0.05f) // 5% threshold
            return false;

        // Don't spam notifications
        string notificationKey = $"stat_{args.CharacterId}_{args.StatType}";
        if (_activeNotifications.Contains(notificationKey))
            return false;

        return true;
    }

    private NotificationImportance GetStatChangeImportance(StatChangeEventArgs args)
    {
        float changePercent = (args.NewValue - args.OldValue) / args.OldValue;

        return changePercent switch
        {
            > 1.0f => NotificationImportance.High,   // Over 100% increase
            > 0.5f => NotificationImportance.Medium, // Over 50% increase
            _ => NotificationImportance.Low
        };
    }
```

```csharp
public class CombatStatIntegration
{
    private readonly CharacterStatSystem _statSystem;

        _activeCombatEffects = new HashSet<string>();
    }

    public void ApplyCombatEffect(CombatEffect effect)
    {
        if (_activeCombatEffects.Contains(effect.Id))
            return;

        var modifiers = CreateCombatModifiers(effect);
        foreach (var modifier in modifiers)
        {
            _statSystem.AddModifier(
                modifier.StatType,
                modifier.Id,
                modifier
            );
        }

        _combatModifiers[effect.Id] = modifiers;
        _activeCombatEffects.Add(effect.Id);
    }

    private List<StatModifier> CreateCombatModifiers(CombatEffect effect)
    {
        var modifiers = new List<StatModifier>();

        switch (effect.Type)
        {
            case CombatEffectType.Buff:
                modifiers.Add(new StatModifier(
                    $"combat_buff_{effect.Id}",
                    "Combat",
                    effect.AffectedStat,
                    BuffType.Percentage,
                    effect.Value,
                    effect.Duration
                ));
                break;

            case CombatEffectType.Debuff:
                modifiers.Add(new StatModifier(
                    $"combat_debuff_{effect.Id}",
                    "Combat",
                    effect.AffectedStat,
                    BuffType.Percentage,
                    -effect.Value,
                    effect.Duration
                ));
                break;

            case CombatEffectType.MultiStat:
                foreach (var statEffect in effect.StatEffects)
                {
                    modifiers.Add(new StatModifier(
                        $"combat_multi_{effect.Id}_{statEffect.StatType}",
                        "Combat",
                        statEffect.StatType,
                        statEffect.Type,
                        statEffect.Value,
                        effect.Duration
                    ));
                }
                break;
        }

        return modifiers;
    }

    public void RemoveCombatEffect(string effectId)
    {
        if (!_combatModifiers.TryGetValue(effectId, out var modifiers))
            return;

        foreach (var modifier in modifiers)
        {
            _statSystem.RemoveModifier(modifier.StatType, modifier.Id);
        }

        _combatModifiers.Remove(effectId);
        _activeCombatEffects.Remove(effectId);
```

### B. Performance Optimization

```csharp
public class StatCalculationCache
{
    private readonly struct CacheKey
    {
        public readonly StatType StatType;
        public readonly int ModifierHash;
        public readonly int BonusHash;

        public CacheKey(
            StatType statType, 
            IEnumerable<StatModifier> modifiers,
            IEnumerable<PermanentBonus> bonuses)
        {
            StatType = statType;
            ModifierHash = CalculateModifierHash(modifiers);
            BonusHash = CalculateBonusHash(bonuses);
        }

        private static int CalculateModifierHash(IEnumerable<StatModifier> modifiers)
        {
            int hash = 17;
            foreach (var mod in modifiers.OrderBy(m => m.Id))
            {
                hash = hash * 31 + mod.GetHashCode();
            }
            return hash;
        }

        private static int CalculateBonusHash(IEnumerable<PermanentBonus> bonuses)
        {
            int hash = 17;
            foreach (var bonus in bonuses.OrderBy(b => b.Id))
            {
                hash = hash * 31 + bonus.GetHashCode();
            }
            return hash;
        }
    }

    private class CacheEntry
    {
        public float Value { get; set; }
        public DateTime Timestamp { get; set; }
    }

    private readonly Dictionary<CacheKey, CacheEntry> _cache;
    private readonly int _maxCacheSize;
    private readonly TimeSpan _cacheTimeout;

    public StatCalculationCache(
        int maxCacheSize = 1000,
        TimeSpan? cacheTimeout = null)
    {
        _cache = new Dictionary<CacheKey, CacheEntry>();
        _maxCacheSize = maxCacheSize;
        _cacheTimeout = cacheTimeout ?? TimeSpan.FromMinutes(1);
    }

    public bool TryGetCachedValue(
        StatType statType,
        IEnumerable<StatModifier> modifiers,
        IEnumerable<PermanentBonus> bonuses,
        out float value)
    {
        var key = new CacheKey(statType, modifiers, bonuses);
        
        if (_cache.TryGetValue(key, out var entry))
        {
            if (DateTime.UtcNow - entry.Timestamp <= _cacheTimeout)
            {
                value = entry.Value;
                return true;
            }
            
            _cache.Remove(key);
        }

        value = 0f;
        return false;
    }

    public void CacheValue(
        StatType statType,
        IEnumerable<StatModifier> modifiers,
        IEnumerable<PermanentBonus> bonuses,
        float value)
    {
        var key = new CacheKey(statType, modifiers, bonuses);
        
        if (_cache.Count >= _maxCacheSize)
        {
            // Remove oldest entries
            var oldestKeys = _cache
                .OrderBy(kvp => kvp.Value.Timestamp)
                .Take(_cache.Count - _maxCacheSize + 1)
                .Select(kvp => kvp.Key)
                .ToList();

            foreach (var oldKey in oldestKeys)
            {
                _cache.Remove(oldKey);
            }
        }

        _cache[key] = new CacheEntry
        {
            Value = value,
```

```csharp
public partial class CharacterStatSystem
{
    private readonly StatCalculationCache _calculationCache;
    private readonly ConcurrentDictionary<StatType, object> _calculationLocks;

    public CharacterStatSystem(string characterId)
    {
        // ... existing initialization ...
        _calculationCache = new StatCalculationCache();
        _calculationLocks = new ConcurrentDictionary<StatType, object>();
    }

    public float GetStatValue(StatType statType)
    {
        // Try cache first
        if (_calculationCache.TryGetCachedValue(
            statType,
            GetModifiers(statType),
            GetPermanentBonuses(statType),
            out float cachedValue))
        {
            return cachedValue;
        }

        // Get or create lock for this stat type
        var lockObj = _calculationLocks.GetOrAdd(
            statType,
            _ => new object()
        );

        // Perform thread-safe calculation
        lock (lockObj)
        {
            float value = CalculateStatValue(statType);
            
            // Cache the result
            _calculationCache.CacheValue(
                statType,
                GetModifiers(statType),
                GetPermanentBonuses(statType),
                value
            );
            
            return value;
        }
    }

    private float CalculateStatValue(StatType statType)
    {
        if (!_stats.TryGetValue(statType, out var stat))
            return 0f;

        var (flatBonus, percentageBonus) = _modifierManager
            .CalculateModifiers(statType);

        float baseWithFlat = stat.BaseValue + flatBonus;
        float withPercentage = baseWithFlat * (1 + percentageBonus);

        // Apply permanent bonuses
        float permanentBonus = _bonusRegistry
            .GetTotalBonusForStat(statType);
        
        return withPercentage * (1 + permanentBonus);
    }

    public void InvalidateCache(StatType? statType = null)
    {
        if (statType.HasValue)
        {
            // Invalidate specific stat
            var lockObj = _calculationLocks.GetOrAdd(
                statType.Value,
                _ => new object()
            );

            lock (lockObj)
            {
                _calculationCache.InvalidateCache();
            }
        }
        else
        {
            // Invalidate all stats
            _calculationCache.InvalidateCache();
        }
    }
}
```

```csharp
public class MilestoneBonusManager
{
    private readonly string _characterId;
    private readonly BonusRegistry _bonusRegistry;
    private readonly Dictionary<string, MilestoneProgress> _milestoneProgress;
    private readonly List<MilestoneDefinition> _milestoneDefinitions;

    public MilestoneBonusManager(
        string characterId,
        BonusRegistry bonusRegistry)
    {
        _characterId = characterId;
        _bonusRegistry = bonusRegistry;
        _milestoneProgress = new Dictionary<string, MilestoneProgress>();
        _milestoneDefinitions = LoadMilestoneDefinitions();
    }

    public void UpdateProgress(string milestoneId, float progress)
    {
        if (!_milestoneProgress.TryGetValue(milestoneId, out var milestoneProgress))
        {
            milestoneProgress = new MilestoneProgress();
            _milestoneProgress[milestoneId] = milestoneProgress;
        }

        float oldProgress = milestoneProgress.CurrentProgress;
        milestoneProgress.CurrentProgress = progress;

        var definition = _milestoneDefinitions.FirstOrDefault(m => m.Id == milestoneId);
        if (definition == null)
            return;

        // Check if any tiers were completed
        foreach (var tier in definition.Tiers.OrderBy(t => t.RequiredProgress))
        {
            if (!milestoneProgress.CompletedTiers.Contains(tier.Level) &&
                progress >= tier.RequiredProgress)
            {
                CompleteMilestoneTier(definition, tier);
                milestoneProgress.CompletedTiers.Add(tier.Level);
            }
        }

        RaiseMilestoneProgressEvent(
            milestoneId, 
            oldProgress, 
            progress,
            definition.MaxProgress
        );
    }

    private void CompleteMilestoneTier(
        MilestoneDefinition definition,
        MilestoneTier tier)
    {
        foreach (var reward in tier.Rewards)
        {
            if (reward is StatBonusReward statReward)
            {
                var bonus = new MilestoneBonus(
                    $"{definition.Id}_tier_{tier.Level}",
                    $"{definition.Name} Tier {tier.Level}",
                    statReward.StatType,
                    MilestoneType.Achievement,
                    statReward.BonusValue
                );

                _bonusRegistry.AddBonus(bonus);
            }
        }

        RaiseMilestoneTierCompletedEvent(definition.Id, tier.Level);
    }

    private void RaiseMilestoneProgressEvent(
        string milestoneId,
        float oldProgress,
        float newProgress,
        float maxProgress)
    {
        Globals.Instance?.gameMangers?.Events?.RaiseEvent(
            EventType.MilestoneProgress,
            new MilestoneProgressEventArgs(
                _characterId,
                milestoneId,
                oldProgress,
                newProgress,
                maxProgress
            )
        );
    }

    private void RaiseMilestoneTierCompletedEvent(
        string milestoneId,
        int tier)
    {
        Globals.Instance?.gameMangers?.Events?.RaiseEvent(
            EventType.MilestoneTierCompleted,
            new MilestoneTierEventArgs(
                _characterId,
                milestoneId,
                tier
            )
        );
    }
}
```

```csharp
public class StatDebugger

    private readonly CharacterStatSystem _statSystem;
    
    public StatDebugger(CharacterStatSystem statSystem)

    }

    public string GetStatBreakdown(StatType statType)
    {
        var stat = _statSystem.GetStat(statType);
        if (stat == null)
            return $"No stat found for {statType}";

        var breakdown = new StringBuilder();
        breakdown.AppendLine($"=== {statType} Breakdown ===");
        breakdown.AppendLine($"Base Value: {stat.BaseValue:F2}");

        // Get modifiers
        var (flatBonus, percentageBonus) = _statSystem._modifierManager.CalculateModifiers(statType);
        breakdown.AppendLine($"Flat Modifiers: +{flatBonus:F2}");
        breakdown.AppendLine($"Percentage Modifiers: +{percentageBonus:P2}");

        // Get permanent bonuses
        var permanentBonuses = _statSystem.GetPermanentBonuses(statType);
        if (permanentBonuses.Any())
        {
            breakdown.AppendLine("\nPermanent Bonuses:");
            foreach (var bonus in permanentBonuses)
            {
                breakdown.AppendLine($"- {bonus.Name}: +{bonus.Value:P2}");
            }
        }

        // Final calculation
        float finalValue = _statSystem.GetStatValue(statType);
        breakdown.AppendLine($"\nFinal Value: {finalValue:F2}");

        return breakdown.ToString();
    }

    public string GetModifierStackInspection(StatType statType)
    {
        var modifiers = _statSystem._modifierManager.GetModifiersForStat(statType);
        if (!modifiers.Any())
            return $"No active modifiers for {statType}";

        var inspection = new StringBuilder();
        inspection.AppendLine($"=== {statType} Modifier Stack ===");
        
        // Group by modifier type
        var grouped = modifiers
            .GroupBy(m => m.Type)
            .OrderBy(g => g.Key);

        foreach (var group in grouped)
        {
            inspection.AppendLine($"\n{group.Key} Modifiers:");
            foreach (var mod in group.OrderBy(m => m.Source))
            {
                var timeLeft = mod.Duration.HasValue ? 
                    (mod.StartTime + mod.Duration.Value - DateTime.UtcNow).TotalSeconds : 
                    double.PositiveInfinity;
                
                inspection.AppendLine(
                    $"- {mod.Source}: {mod.Value:F2} " +
                    $"(ID: {mod.Id}, Time Left: {(timeLeft == double.PositiveInfinity ? "Permanent" : $"{timeLeft:F1}s")})");
            }
        }

        return inspection.ToString();
    }

    public string GetBonusCalculationTrace(StatType statType)
    {
        var trace = new StringBuilder();
        trace.AppendLine($"=== {statType} Bonus Calculation Trace ===");

        var stat = _statSystem.GetStat(statType);
        float baseValue = stat.BaseValue;
        trace.AppendLine($"Starting with base value: {baseValue:F2}");

        // Trace flat modifiers
        var (flatBonus, percentageBonus) = _statSystem._modifierManager.CalculateModifiers(statType);
        if (flatBonus != 0)
        {
            float afterFlat = baseValue + flatBonus;
            trace.AppendLine($"Adding flat modifiers (+{flatBonus:F2}): {afterFlat:F2}");
            baseValue = afterFlat;

        if (percentageBonus != 0)
        {
            float afterPercentage = baseValue * (1 + percentageBonus);
            trace.AppendLine($"Applying percentage modifiers (+{percentageBonus:P2}): {afterPercentage:F2}");
            baseValue = afterPercentage;
        }

        // Trace permanent bonuses
        var permanentBonuses = _statSystem.GetPermanentBonuses(statType);
        foreach (var bonus in permanentBonuses.OrderByDescending(b => b.StackPriority))
        {
            float afterBonus = bonus.CalculateBonus(baseValue);
            trace.AppendLine($"Applying {bonus.Name} ({bonus.Type}): {afterBonus:F2}");
            baseValue = afterBonus;
        }

        trace.AppendLine($"\nFinal Value: {baseValue:F2}");
        return trace.ToString();
    }
}
```

```csharp
public class StatComparisonAnalyzer
{
    public class StatComparison
    {
        public StatType StatType { get; set; }
        public float FirstValue { get; set; }
        public float SecondValue { get; set; }
        public float Difference { get; set; }
        public float PercentageDifference { get; set; }
        public Dictionary<string, float> ModifierComparison { get; set; }
        public Dictionary<string, float> BonusComparison { get; set; }
    }

    public string CompareCharacters(Character first, Character second)
    {
        var output = new StringBuilder();
        output.AppendLine($"=== Character Comparison ===");
        output.AppendLine($"{first.Name} vs {second.Name}\n");

        var comparisons = new Dictionary<StatType, StatComparison>();

        // Compare each stat type
        foreach (StatType statType in Enum.GetValues(typeof(StatType)))
        {
            var comparison = CompareStats(first, second, statType);
            comparisons[statType] = comparison;
        }

        // Group by significance
        var significantDiffs = comparisons
            .Where(c => Math.Abs(c.Value.PercentageDifference) >= 0.1f) // 10% or more
            .OrderByDescending(c => Math.Abs(c.Value.PercentageDifference));

        output.AppendLine("Significant Differences:");
        foreach (var diff in significantDiffs)
        {
            var comp = diff.Value;
            output.AppendLine(
                $"{comp.StatType}:\n" +
                $"  {first.Name}: {comp.FirstValue:F2}\n" +
                $"  {second.Name}: {comp.SecondValue:F2}\n" +
                $"  Difference: {(comp.Difference >= 0 ? "+" : "")}{comp.Difference:F2} ({comp.PercentageDifference:P2})"
            );

            // Show modifier differences
            if (comp.ModifierComparison.Any())
            {
                output.AppendLine("  Modifier Differences:");
                foreach (var mod in comp.ModifierComparison)
                {
                    output.AppendLine($"    {mod.Key}: {(mod.Value >= 0 ? "+" : "")}{mod.Value:P2}");
                }
            }
        }

        return output.ToString();
    }

    private StatComparison CompareStats(Character first, Character second, StatType statType)
    {
        float firstValue = first.GetStatValue(statType);
        float secondValue = second.GetStatValue(statType);

        var comparison = new StatComparison
        {
            StatType = statType,
            FirstValue = firstValue,
            SecondValue = secondValue,
            Difference = secondValue - firstValue,
            PercentageDifference = (secondValue - firstValue) / firstValue,
            ModifierComparison = CompareModifiers(first, second, statType),
            BonusComparison = CompareBonuses(first, second, statType)
        };

        return comparison;
    }

    private Dictionary<string, float> CompareModifiers(
        Character first, 
        Character second, 
        StatType statType)
    {
        var differences = new Dictionary<string, float>();
        var firstMods = first._statSystem._modifierManager.GetModifiersForStat(statType)
            .GroupBy(m => m.Source)
            .ToDictionary(g => g.Key, g => g.Sum(m => m.Value));
        var secondMods = second._statSystem._modifierManager.GetModifiersForStat(statType)
            .GroupBy(m => m.Source)
            .ToDictionary(g => g.Key, g => g.Sum(m => m.Value));

        foreach (var source in firstMods.Keys.Union(secondMods.Keys))
        {
            float firstValue = firstMods.GetValueOrDefault(source);
            float secondValue = secondMods.GetValueOrDefault(source);
            if (Math.Abs(firstValue - secondValue) > 0.001f)
            {
                differences[source] = secondValue - firstValue;
            }
        }

        return differences;
    }
}

public class StatBenchmarkTool
{
    public class StatBenchmark
    {
        public StatType StatType { get; set; }
        public float MinValue { get; set; }
        public float MaxValue { get; set; }
        public float AverageValue { get; set; }
        public float Median { get; set; }
        public Dictionary<int, float> Percentiles { get; set; }
        public float StandardDeviation { get; set; }
    }

    public string BenchmarkStatType(StatType statType, IEnumerable<Character> characters)
    {
        var values = characters
            .Select(c => c.GetStatValue(statType))
            .OrderBy(v => v)
            .ToList();

        if (!values.Any())
            return $"No data available for {statType}";

        var benchmark = new StatBenchmark
        {
            StatType = statType,
            MinValue = values.First(),
            MaxValue = values.Last(),
            AverageValue = values.Average(),
            Median = CalculateMedian(values),
            Percentiles = CalculatePercentiles(values),
            StandardDeviation = CalculateStandardDeviation(values)
        };

        return FormatBenchmarkReport(benchmark);
    }

    public string PowerLevelAnalysis(
        IEnumerable<Character> characters, 
        Dictionary<StatType, PowerWeight> weights)
    {
        var output = new StringBuilder();
        output.AppendLine("=== Power Level Analysis ===\n");

        // Calculate weighted power levels
        var powerLevels = characters
            .Select(c => new
            {
                Character = c,
                PowerLevel = CalculateWeightedPower(c, weights)
            })
            .OrderByDescending(x => x.PowerLevel)
            .ToList();

        // Calculate statistics
        float averagePower = powerLevels.Average(x => x.PowerLevel);
        float standardDev = CalculateStandardDeviation(
            powerLevels.Select(x => x.PowerLevel));

        output.AppendLine($"Average Power Level: {averagePower:F2}");
        output.AppendLine($"Standard Deviation: {standardDev:F2}\n");

        // Identify outliers
        output.AppendLine("Power Level Distribution:");
        foreach (var entry in powerLevels)
        {
            float zScore = (entry.PowerLevel - averagePower) / standardDev;
            string outlierMark = Math.Abs(zScore) > 2 ? " <!>" : "";
            
            output.AppendLine(
                $"{entry.Character.Name}: {entry.PowerLevel:F2} " +
                $"(z-score: {zScore:F2}){outlierMark}"
            );
        }

        return output.ToString();
    }

    private float CalculateWeightedPower(
        Character character, 
        Dictionary<StatType, PowerWeight> weights)
    {
        float power = 0f;
        foreach (var (statType, weight) in weights)
        {
            float value = character.GetStatValue(statType);
            power += value * weight.Value * weight.Scaling;
        }
        return power;
    }

    private string FormatBenchmarkReport(StatBenchmark benchmark)
    {
        var output = new StringBuilder();
        output.AppendLine($"=== Benchmark Report for {benchmark.StatType} ===\n");

        output.AppendLine($"Range: {benchmark.MinValue:F2} - {benchmark.MaxValue:F2}");
        output.AppendLine($"Average: {benchmark.AverageValue:F2}");
        output.AppendLine($"Median: {benchmark.Median:F2}");
        output.AppendLine($"Standard Deviation: {benchmark.StandardDeviation:F2}\n");

        output.AppendLine("Percentiles:");
        foreach (var percentile in benchmark.Percentiles.OrderBy(p => p.Key))
        {
            output.AppendLine($"{percentile.Key}th: {percentile.Value:F2}");
        }

        return output.ToString();
    }

    private float CalculateMedian(List<float> values)
    {
        int mid = values.Count / 2;
        if (values.Count % 2 == 0)
            return (values[mid - 1] + values[mid]) / 2;
        return values[mid];
    }

    private Dictionary<int, float> CalculatePercentiles(List<float> values)
    {
        return new[] { 10, 25, 75, 90 }
            .ToDictionary(
                p => p,
                p => CalculatePercentile(values, p)
            );
    }

    private float CalculatePercentile(List<float> values, int percentile)
    {
        int index = (int)Math.Round(values.Count * (percentile / 100f)) - 1;
        return values[Math.Max(0, Math.Min(index, values.Count - 1))];
    }

    private float CalculateStandardDeviation(IEnumerable<float> values)
    {
        float average = values.Average();
        float sumOfSquaresOfDifferences = values.Sum(v => (v - average) * (v - average));
        return (float)Math.Sqrt(sumOfSquaresOfDifferences / values.Count());
    }
}

    }
```

public abstract class BaseMilestone
{
    public string Id { get; }
    public string Name { get; }
    public MilestoneType Type { get; }
    public bool IsRepeatable { get; }
    public MilestoneProgress Progress { get; private set; }
    public List<MilestoneReward> Rewards { get; }
    public DateTime CreatedTime { get; }

    protected BaseMilestone(
        string name,
        MilestoneType type,
        List<MilestoneReward> rewards,
        bool isRepeatable = false)
    {
        Id = Guid.NewGuid().ToString();
        Name = name;
        Type = type;
        Rewards = rewards;
        IsRepeatable = isRepeatable;
        Progress = new MilestoneProgress();
        CreatedTime = DateTime.UtcNow;
    }

    /// <summary>
    /// Abstract method for checking if the milestone is completed
    /// </summary>
    public abstract bool CheckCompletion(string characterId);

    /// <summary>
    /// Abstract method for updating milestone progress
    /// </summary>
    public abstract void UpdateProgress(string characterId, float value);

    /// <summary>
    /// Completes the milestone and applies rewards
    /// </summary>
    public virtual void Complete(string characterId)
    {
        if (Progress.IsCompleted && !IsRepeatable)
            return;

        Progress.CompleteCount++;
        Progress.LastCompletionTime = DateTime.UtcNow;

        foreach (var reward in Rewards)
        {
            reward.Apply(characterId);
        }

        RaiseMilestoneCompletedEvent(characterId);
    }

    /// <summary>
    /// Resets milestone progress
    /// </summary>
    public virtual void Reset()
    {
        Progress = new MilestoneProgress();
    }

    /// <summary>
    /// Gets the current completion percentage
    /// </summary>
    public abstract float GetCompletionPercentage();

    protected void RaiseMilestoneCompletedEvent(string characterId)
    {
        Globals.Instance?.gameMangers?.Events?.RaiseEvent(
            EventType.MilestoneCompleted,
            new MilestoneEventArgs(
                characterId,
                Id,
                Type,
                Progress.CompleteCount,
                Rewards.Select(r => r.Description).ToList()
            )
        );
    }

    protected void RaiseMilestoneProgressEvent(string characterId, float oldProgress, float newProgress)
    {
        Globals.Instance?.gameMangers?.Events?.RaiseEvent(
            EventType.MilestoneProgress,
            new MilestoneProgressEventArgs(
                characterId,
                Id,
                Name,
                oldProgress,
                newProgress,
                GetCompletionPercentage()
            )
        );
    }
}

public class MilestoneProgress
{
    public int CompleteCount { get; set; }
    public DateTime? LastCompletionTime { get; set; }
    public float CurrentValue { get; set; }
    public bool IsCompleted => CompleteCount > 0;
}

public class StatMilestone : BaseMilestone
{
    public StatType StatType { get; }
    public float TargetValue { get; }

    public StatMilestone(
        string name,
        StatType statType,
        float targetValue,
        List<MilestoneReward> rewards,
        bool isRepeatable = false)
        : base(name, MilestoneType.Stat, rewards, isRepeatable)
    {
        StatType = statType;
        TargetValue = targetValue;
    }

    public override bool CheckCompletion(string characterId)
    {
        var character = Globals.Instance.gameMangers.Player.GetCharacter(characterId);
        float currentValue = character.GetStatValue(StatType);
        return currentValue >= TargetValue;
    }

    public override void UpdateProgress(string characterId, float value)
    {
        float oldValue = Progress.CurrentValue;
        Progress.CurrentValue = value;

        RaiseMilestoneProgressEvent(characterId, oldValue, value);

        if (CheckCompletion(characterId))
        {
            Complete(characterId);
        }
    }

    public override float GetCompletionPercentage()
    {
        return Math.Min(1f, Progress.CurrentValue / TargetValue);
    }
}
