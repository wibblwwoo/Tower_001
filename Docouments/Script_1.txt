using Godot;

public partial class CustomProgressBar : Control
{
    [Export]
    public float Value { get; set; } = 0;
    
    private ColorRect _background;
    private ColorRect _fill;
    
    // Default colors
    private readonly Color DEFAULT_BLUE = Colors.Blue;
    private readonly Color DEFAULT_GREEN = Colors.Green;
    private readonly Color DEFAULT_RED = Colors.Red;
    private readonly Color DEFAULT_YELLOW = Colors.Yellow;
    private readonly Color DEFAULT_BACKGROUND = Colors.Gray;
    
    // Current colors - these can be modified during runtime
    [Export]
    public Color BlueStageColor { get; set; }
    [Export]
    public Color GreenStageColor { get; set; }
    [Export]
    public Color RedStageColor { get; set; }
    [Export]
    public Color YellowStageColor { get; set; }
    [Export]
    public Color BackgroundColor { get; set; }
    
    public override void _Ready()
    {
        ResetColors();
        
        CustomMinimumSize = new Vector2(200, 20);
        
        _background = new ColorRect
        {
            Color = BackgroundColor,
            CustomMinimumSize = CustomMinimumSize
        };
        AddChild(_background);
        
        _fill = new ColorRect
        {
            Color = BlueStageColor,
            CustomMinimumSize = new Vector2(0, CustomMinimumSize.Y)
        };
        AddChild(_fill);
    }
    
    public override void _Process(double delta)
    {
        _fill.Color = GetColorForProgress(Value);
        _fill.Size = new Vector2((Value / 100) * Size.X, Size.Y);
        _background.Color = BackgroundColor;
        CheckProgress();
    }
    
    /// <summary>
    /// Resets all colors back to their default values
    /// </summary>
    public void ResetColors()
    {
        BlueStageColor = DEFAULT_BLUE;
        GreenStageColor = DEFAULT_GREEN;
        RedStageColor = DEFAULT_RED;
        YellowStageColor = DEFAULT_YELLOW;
        BackgroundColor = DEFAULT_BACKGROUND;
    }
    
    /// <summary>
    /// Resets the progress value back to 0
    /// </summary>
    public void ResetProgress()
    {
        Value = 0;
    }
    
    /// <summary>
    /// Resets both colors and progress to default values
    /// </summary>
    public void ResetAll()
    {
        ResetColors();
        ResetProgress();
    }
    
    /// <summary>
    /// Sets custom colors for all stages at once
    /// </summary>
    public void SetCustomColors(Color blue, Color green, Color red, Color yellow, Color background)
    {
        BlueStageColor = blue;
        GreenStageColor = green;
        RedStageColor = red;
        YellowStageColor = yellow;
        BackgroundColor = background;
    }
    
    private Color GetColorForProgress(float progress)
    {
        if (progress <= 25)
        {
            return LerpColor(BlueStageColor, GreenStageColor, progress / 25f);
        }
        else if (progress <= 50)
        {
            return LerpColor(GreenStageColor, RedStageColor, (progress - 25) / 25f);
        }
        else if (progress <= 75)
        {
            return LerpColor(RedStageColor, YellowStageColor, (progress - 50) / 25f);
        }
        else
        {
            return YellowStageColor;
        }
    }
    
    private Color LerpColor(Color fromColor, Color toColor, float weight)
    {
        weight = Mathf.Clamp(weight, 0, 1);
        
        return new Color(
            Mathf.Lerp(fromColor.R, toColor.R, weight),
            Mathf.Lerp(fromColor.G, toColor.G, weight),
            Mathf.Lerp(fromColor.B, toColor.B, weight),
            Mathf.Lerp(fromColor.A, toColor.A, weight)
        );
    }
    
    private void CheckProgress()
    {
        if (Value == 0) OnZeroProgress();
        else if (Value == 25) OnQuarterProgress();
        else if (Value == 50) OnHalfProgress();
        else if (Value == 75) OnThreeQuarterProgress();
        else if (Value == 100) OnFullProgress();
    }
    
    private void OnZeroProgress() 
    { 
        // Event handling for 0%
    }
    
    private void OnQuarterProgress() 
    { 
        // Event handling for 25%
    }
    
    private void OnHalfProgress() 
    { 
        // Event handling for 50%
    }
    
    private void OnThreeQuarterProgress() 
    { 
        // Event handling for 75%
    }
    
    private void OnFullProgress() 
    { 
        // Event handling for 100%
    }
}

    private Panel _rightPanel;
    private VBoxContainer _leftContainer;
    private VBoxContainer _rightContainer;

    {
        // Make the control fill the window
        AnchorsPreset = (int)LayoutPreset.FullRect;

        _mainContainer = new HBoxContainer
        {
            AnchorsPreset = (int)LayoutPreset.FullRect
        };
        
        // Left panel setup
        _leftPanel = new Panel
        {
            CustomMinimumSize = new Vector2(200, 0)
        };
        
        _leftContainer = new VBoxContainer
        {
            AnchorsPreset = (int)LayoutPreset.FullRect
        };
        _leftPanel.AddChild(_leftContainer);
        
        // Right panel setup
        _rightPanel = new Panel();
        _rightContainer = new VBoxContainer
        {
            AnchorsPreset = (int)LayoutPreset.FullRect
        };
        _rightPanel.AddChild(_rightContainer);
        
        // Add panels to main container
        _mainContainer.AddChild(_leftPanel);
        _mainContainer.AddChild(_rightPanel);
        
        // Set size flags for scaling
        _leftPanel.SizeFlagsHorizontal = Control.SizeFlags.Expand | Control.SizeFlags.Fill;
        _rightPanel.SizeFlagsHorizontal = Control.SizeFlags.Expand | Control.SizeFlags.Fill;
        
        // Set the stretch ratio for 25/75 split
        _leftPanel.SizeFlagsStretchRatio = 25;
        _rightPanel.SizeFlagsStretchRatio = 75;
        
        AddChild(_mainContainer);
        
        // Set panel colors
        _leftPanel.AddThemeColorOverride("panel", new Color(0.2f, 0.2f, 0.3f));
        _rightPanel.AddThemeColorOverride("panel", new Color(0.3f, 0.2f, 0.2f));
        
        // Add some padding to the containers
        _leftContainer.AddThemeConstantOverride("separation", 10);
        _rightContainer.AddThemeConstantOverride("separation", 10);
        
        // Set custom margins for the containers
        var margin = new Margin(10, 10, 10, 10);
        _leftContainer.Position = margin.Position;
        _rightContainer.Position = margin.Position;
    }

        _leftContainer.AddChild(node);
    }

        _rightContainer.AddChild(node);
    }

        AddToLeftPanel(leftLabel);

# Managers/UIAnimationManager.cs
public class UIAnimationManager : Node
{
    private Dictionary<IUITransitionable, Tween> _activeTweens = new Dictionary<IUITransitionable, Tween>();
    
    public async Task AnimateShow(Control control, TransitionSettings settings)
    {
        if (control is IUITransitionable transitionable)
        {
            // Stop any active tween
            if (_activeTweens.ContainsKey(transitionable))
            {
                _activeTweens[transitionable].Kill();
                _activeTweens.Remove(transitionable);
            }
            
            control.Visible = true;
            var tween = CreateTween();
            _activeTweens[transitionable] = tween;
            
            if (settings.UseAlpha)
            {
                control.Modulate = new Color(1, 1, 1, 0);
                tween.TweenProperty(control, "modulate:a", 1.0f, settings.Duration)
                     .SetTrans(settings.TransitionType)
                     .SetEase(settings.EaseType);
            }
            
            if (settings.UseScale)
            {
                control.Scale = Vector2.Zero;
                tween.TweenProperty(control, "scale", Vector2.One, settings.Duration)
                     .SetTrans(settings.TransitionType)
                     .SetEase(settings.EaseType);
            }
            
            await ToSignal(tween, "finished");
        }
    }
    
    public async Task AnimateHide(Control control, TransitionSettings settings)
    {
        if (control is IUITransitionable transitionable)
        {
            var tween = CreateTween();
            _activeTweens[transitionable] = tween;
            
            if (settings.UseAlpha)
            {
                tween.TweenProperty(control, "modulate:a", 0.0f, settings.Duration)
                     .SetTrans(settings.TransitionType)
                     .SetEase(settings.EaseType);
            }
            
            if (settings.UseScale)
            {
                tween.TweenProperty(control, "scale", Vector2.Zero, settings.Duration)
                     .SetTrans(settings.TransitionType)
                     .SetEase(settings.EaseType);
            }
            
            await ToSignal(tween, "finished");
            control.Visible = false;
        }
    }
}

public class UIHistoryManager : Node
{
    private Stack<string> _stateHistory = new Stack<string>();
    private string _currentState;
    
    public void PushState(string state)
    {
        if (_currentState != null)
        {
            _stateHistory.Push(_currentState);
        }
        _currentState = state;
    }
    
    public string PopState()
    {
        if (_stateHistory.Count > 0)
        {
            _currentState = _stateHistory.Pop();
            return _currentState;
        }
        return null;
    }
    
    public void ClearHistory()
    {
        _stateHistory.Clear();
    }
}

# Managers/UILayerManager.cs
public class UILayerManager : Node
{
    public enum UILayer
    {
        Background,
        Main,
        Overlay,
        Popup,
        Modal
    }
    
    private Dictionary<UILayer, CanvasLayer> _layers = new Dictionary<UILayer, CanvasLayer>();
    
    public override void _Ready()
    {
        foreach (UILayer layer in Enum.GetValues(typeof(UILayer)))
        {
            var canvasLayer = new CanvasLayer
            {
                Layer = (int)layer
            };
            AddChild(canvasLayer);
            _layers[layer] = canvasLayer;
        }
    }
    
    public void AddToLayer(Control control, UILayer layer)
    {
        if (_layers.TryGetValue(layer, out var canvasLayer))
        {
            if (control.GetParent() != null)
            {
                control.GetParent().RemoveChild(control);
            }
            canvasLayer.AddChild(control);
        }
    }
}

# Extensions/ResponsiveContainer.cs
public partial class ResponsiveContainer : Container
{
    public enum BreakPoint
    {
        Small,
        Medium,
        Large
    }
    
    [Export]
    public Vector2 SmallBreakpoint { get; set; } = new Vector2(0, 640);
    
    [Export]
    public Vector2 MediumBreakpoint { get; set; } = new Vector2(641, 1200);
    
    [Export]
    public Vector2 LargeBreakpoint { get; set; } = new Vector2(1201, float.MaxValue);
    
    private BreakPoint _currentBreakpoint = BreakPoint.Medium;
    
    [Signal]
    public delegate void BreakpointChangedEventHandler(BreakPoint newBreakpoint);
    
    public override void _Process(double delta)
    {
        UpdateBreakpoint();
    }
    
    private void UpdateBreakpoint()
    {
        var width = GetViewport().GetVisibleRect().Size.X;
        var newBreakpoint = GetBreakpointForWidth(width);
        
        if (newBreakpoint != _currentBreakpoint)
        {
            _currentBreakpoint = newBreakpoint;
            EmitSignal(SignalName.BreakpointChanged, _currentBreakpoint);
            QueueSort();
        }
    }
    
    private BreakPoint GetBreakpointForWidth(float width)
    {
        if (width <= SmallBreakpoint.Y) return BreakPoint.Small;
        if (width <= MediumBreakpoint.Y) return BreakPoint.Medium;
        return BreakPoint.Large;
    }
}

    public void Subscribe(string eventName, Action<object> handler)

        if (!_eventHandlers.ContainsKey(eventName))
        {
            _eventHandlers[eventName] = new List<Action<object>>();
        }
        _eventHandlers[eventName].Add(handler);
    }
    
    public void Unsubscribe(string eventName, Action<object> handler)
    {
        if (_eventHandlers.ContainsKey(eventName))
        {
            _eventHandlers[eventName].Remove(handler);
        }
    }
    
    public void RaiseEvent(string eventName, object data = null)
    {
        if (_eventHandlers.ContainsKey(eventName))
        {
            foreach (var handler in _eventHandlers[eventName])
            {
                handler.Invoke(data);
            }
        }
    }
}

    {
        base._Ready();
        
        _historyManager = new UIHistoryManager();
        _layerManager = new UILayerManager();
        _animationManager = new UIAnimationManager();
        _eventManager = new UIEventManager();
        
        AddChild(_historyManager);
        AddChild(_layerManager);
        AddChild(_animationManager);
        AddChild(_eventManager);
    }
    
    public async Task ChangeStateAsync(string newState)
    {
        EmitSignal(SignalName.BeforeStateChange, CurrentState, newState);
        _eventManager.RaiseEvent("StateChanging", new { OldState = CurrentState, NewState = newState });
        
        var hideTask = HideCurrentState();
        await hideTask;
        
        _historyManager.PushState(newState);
        CurrentState = newState;
        
        var showTask = ShowNewState(newState);
        await showTask;
        
        _eventManager.RaiseEvent("StateChanged", newState);
    }
    
    public async Task GoBack()
    {
        var previousState = _historyManager.PopState();
        if (previousState != null)
        {
            await ChangeStateAsync(previousState);
        }
    }
    
    private async Task HideCurrentState()
    {
        var hideTasks = new List<Task>();
        
        foreach (var control in GetCurrentStateControls())
        {
            if (control is IUITransitionable transitionable)
            {
                hideTasks.Add(_animationManager.AnimateHide(control as Control, transitionable.TransitionSettings));
            }
            else
            {
                control.Hide();
            }
        }
        
        await Task.WhenAll(hideTasks);
    }
    
    private async Task ShowNewState(string newState)
    {
        var showTasks = new List<Task>();
        
        foreach (var control in GetStateControls(newState))
        {
            if (control is IUITransitionable transitionable)
            {
                showTasks.Add(_animationManager.AnimateShow(control as Control, transitionable.TransitionSettings));
            }
            else
            {
                control.Show();
            }
        }
        
        await Task.WhenAll(showTasks);
    }
}

# Managers/UILayerManager.cs
public class UILayerManager : Node
{
    private Dictionary<UILayerType, CanvasLayer> _layers = new Dictionary<UILayerType, CanvasLayer>();
    private Dictionary<UILayerType, LayerConfiguration> _layerConfigs = new Dictionary<UILayerType, LayerConfiguration>();
    private Dictionary<Control, UILayerType> _controlLayerMap = new Dictionary<Control, UILayerType>();
    
    [Signal]
    public delegate void LayerVisibilityChangedEventHandler(UILayerType layer, bool isVisible);
    [Signal]
    public delegate void LayerInteractionChangedEventHandler(UILayerType layer, bool isInteractable);
    
    public override void _Ready()
    {
        InitializeLayers();
    }
    
    private void InitializeLayers()
    {
        // Create all layers in order
        foreach (UILayerType layerType in Enum.GetValues(typeof(UILayerType)))
        {
            CreateLayer(layerType);
        }
    }
    
    private void CreateLayer(UILayerType layerType)
    {
        var canvasLayer = new CanvasLayer
        {
            Layer = (int)layerType,
            Name = $"Layer_{layerType}"
        };
        
        // Add a Panel as background for the layer
        var background = new Panel
        {
            Name = $"Background_{layerType}",
            MouseFilter = Control.MouseFilterEnum.Ignore
        };
        background.AnchorsPreset = (int)Control.LayoutPreset.FullRect;
        background.Visible = false; // Initially invisible
        
        canvasLayer.AddChild(background);
        AddChild(canvasLayer);
        
        _layers[layerType] = canvasLayer;
        _layerConfigs[layerType] = new LayerConfiguration();
    }
    
    public void AddToLayer(Control control, UILayerType? layerType = null)
    {
        // If no layer specified, try to get it from IUILayerable
        if (!layerType.HasValue && control is IUILayerable layerable)
        {
            layerType = layerable.DefaultLayer;
        }
        
        // Default to Game layer if still no layer specified
        layerType ??= UILayerType.Game;
        
        // Remove from current parent if exists
        if (control.GetParent() != null)
        {
            control.GetParent().RemoveChild(control);
        }
        
        // Add to new layer
        if (_layers.TryGetValue(layerType.Value, out var layer))
        {
            layer.AddChild(control);
            _controlLayerMap[control] = layerType.Value;
            
            // Apply layer configuration
            ApplyLayerConfigToControl(control, layerType.Value);
        }
    }
    
    private void ApplyLayerConfigToControl(Control control, UILayerType layerType)
    {
        if (_layerConfigs.TryGetValue(layerType, out var config))
        {
            control.Visible = config.IsVisible;
            control.MouseFilter = config.IsInteractable ? 
                Control.MouseFilterEnum.Stop : 
                Control.MouseFilterEnum.Ignore;
            
            // Apply opacity
            var modulate = control.Modulate;
            modulate.A = config.Opacity;
            control.Modulate = modulate;
        }
    }
    
    public void ConfigureLayer(UILayerType layerType, LayerConfiguration config)
    {
        _layerConfigs[layerType] = config;
        
        // Update background panel
        if (_layers.TryGetValue(layerType, out var layer))
        {
            var background = layer.GetNode<Panel>($"Background_{layerType}");
            background.Visible = config.BlocksLowerLayers;
            background.Modulate = config.BackgroundTint;
            
            // Update all controls in this layer
            foreach (var control in GetControlsInLayer(layerType))
            {
                ApplyLayerConfigToControl(control, layerType);
            }
        }
    }
    
    public void ShowLayer(UILayerType layerType)
    {
        if (_layerConfigs.TryGetValue(layerType, out var config))
        {
            config.IsVisible = true;
            foreach (var control in GetControlsInLayer(layerType))
            {
                control.Visible = true;
            }
            EmitSignal(SignalName.LayerVisibilityChanged, layerType, true);
        }
    }
    
    public void HideLayer(UILayerType layerType)
    {
        if (_layerConfigs.TryGetValue(layerType, out var config))
        {
            config.IsVisible = false;
            foreach (var control in GetControlsInLayer(layerType))
            {
                control.Visible = false;
            }
            EmitSignal(SignalName.LayerVisibilityChanged, layerType, false);
        }
    }
    
    public void SetLayerInteractable(UILayerType layerType, bool interactable)
    {
        if (_layerConfigs.TryGetValue(layerType, out var config))
        {
            config.IsInteractable = interactable;
            foreach (var control in GetControlsInLayer(layerType))
            {
                control.MouseFilter = interactable ? 
                    Control.MouseFilterEnum.Stop : 
                    Control.MouseFilterEnum.Ignore;
            }
            EmitSignal(SignalName.LayerInteractionChanged, layerType, interactable);
        }
    }
    
    private IEnumerable<Control> GetControlsInLayer(UILayerType layerType)
    {
        return _controlLayerMap.Where(kvp => kvp.Value == layerType)
                              .Select(kvp => kvp.Key);
    }
    
    // Example of how to handle modal windows
    public void ShowModal(Control modalControl)
    {
        // Configure modal layer
        ConfigureLayer(UILayerType.Modal, new LayerConfiguration
        {
            IsVisible = true,
            IsInteractable = true,
            BlocksLowerLayers = true,
            BackgroundTint = new Color(0, 0, 0, 0.5f) // Semi-transparent black
        });
        
        // Disable interaction with lower layers
        foreach (var layerType in Enum.GetValues(typeof(UILayerType)))
        {
            if ((UILayerType)layerType < UILayerType.Modal)
            {
                SetLayerInteractable((UILayerType)layerType, false);
            }
        }
        
        AddToLayer(modalControl, UILayerType.Modal);
    }
    
    public void HideModal()
    {
        // Hide modal layer
        HideLayer(UILayerType.Modal);
        
        // Re-enable interaction with lower layers
        foreach (var layerType in Enum.GetValues(typeof(UILayerType)))
        {
            if ((UILayerType)layerType < UILayerType.Modal)
            {
                SetLayerInteractable((UILayerType)layerType, true);
            }
        }
    }
}
