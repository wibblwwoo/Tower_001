/// <summary>
/// Defines a resource conversion recipe

public class ConversionRecipe
{
    public Dictionary<ResourceType, float> Inputs { get; set; } = new();
    public Dictionary<ResourceType, float> Outputs { get; set; } = new();

}

/// <summary>
/// Manages active resource conversions
/// </summary>
public class ActiveConversion
{
    public ConversionRecipe Recipe { get; set; }
    public float Progress { get; set; } = 0f;
    public bool IsComplete => Progress >= Recipe.ConversionTime;
}

/// <summary>
/// Enhanced resource manager with conversion and storage systems
/// </summary>
public partial class ResourceManager : Node
{
    // Additional signals
    [Signal]
    public delegate void StorageLimitReachedEventHandler(ResourceType type);
    
    [Signal]
    public delegate void ConversionStartedEventHandler(ConversionRecipe recipe);
    
    [Signal]
    public delegate void ConversionCompletedEventHandler(ConversionRecipe recipe);
    
    [Signal]
    public delegate void ConversionProgressUpdatedEventHandler(ConversionRecipe recipe, float progress);

    // Storage and conversion tracking
    private readonly Dictionary<string, ConversionRecipe> _conversionRecipes = new();
    private readonly List<ActiveConversion> _activeConversions = new();
    
    // Called every frame to update conversions
    public override void _Process(double delta)
    {
        UpdateActiveConversions(delta);
    }

    /// <summary>
    /// Updates progress on all active conversions
    /// </summary>
    private void UpdateActiveConversions(double delta)
    {
        for (int i = _activeConversions.Count - 1; i >= 0; i--)
        {
            var conversion = _activeConversions[i];
            conversion.Progress += (float)delta;
            
            EmitSignal(SignalName.ConversionProgressUpdated, 
                conversion.Recipe, 
                conversion.Progress / conversion.Recipe.ConversionTime);

            if (conversion.IsComplete)
            {
                CompleteConversion(conversion);
                _activeConversions.RemoveAt(i);
                
                // Handle auto-repeat if enabled and resources are available
                if (conversion.Recipe.AutoRepeat && CanStartConversion(conversion.Recipe))
                {
                    StartConversion(conversion.Recipe);
                }
            }
        }
    }

    /// <summary>
    /// Registers a new conversion recipe
    /// </summary>
    public void RegisterConversionRecipe(string recipeId, ConversionRecipe recipe)
    {
        _conversionRecipes[recipeId] = recipe;
    }

    /// <summary>
    /// Gets all available conversion recipes
    /// </summary>
    public IEnumerable<KeyValuePair<string, ConversionRecipe>> GetAvailableRecipes()
    {
        return _conversionRecipes.Where(kvp => CanStartConversion(kvp.Value));
    }

    /// <summary>
    /// Checks if a conversion can be started
    /// </summary>
    public bool CanStartConversion(ConversionRecipe recipe)
    {
        // Check if we have enough input resources
        if (!HasSufficientResources(recipe.Inputs))
            return false;

        // Check if we have enough storage space for outputs
        return HasSufficientStorage(recipe.Outputs);
    }

    /// <summary>
    /// Checks if we have sufficient resources for a conversion
    /// </summary>
    private bool HasSufficientResources(Dictionary<ResourceType, float> requirements)
    {
        return requirements.All(req => GetResourceAmount(req.Key) >= req.Value);
    }

    /// <summary>
    /// Checks if we have sufficient storage space for outputs
    /// </summary>
    private bool HasSufficientStorage(Dictionary<ResourceType, float> outputs)
    {
        return outputs.All(output => 
            GetResourceAmount(output.Key) + output.Value <= GetStorageLimit(output.Key));
    }

    /// <summary>
    /// Starts a conversion process
    /// </summary>
    public bool StartConversion(ConversionRecipe recipe)
    {
        if (!CanStartConversion(recipe))
            return false;

        // Consume input resources
        foreach (var (resourceType, amount) in recipe.Inputs)
        {
            RemoveResource(resourceType, amount);
        }

        // Create active conversion
        var conversion = new ActiveConversion { Recipe = recipe };
        _activeConversions.Add(conversion);
        
        EmitSignal(SignalName.ConversionStarted, recipe);
        return true;
    }

    /// <summary>
    /// Completes a conversion and grants output resources
    /// </summary>
    private void CompleteConversion(ActiveConversion conversion)
    {
        foreach (var (resourceType, amount) in conversion.Recipe.Outputs)
        {
            AddResource(resourceType, amount);
        }
        
        EmitSignal(SignalName.ConversionCompleted, conversion.Recipe);
    }

    /// <summary>
    /// Override of AddResource to respect storage limits
    /// </summary>
    public bool AddResource(ResourceType type, float amount)
    {
        float currentAmount = GetResourceAmount(type);
        float limit = GetStorageLimit(type);
        
        // Check if adding would exceed limit
        if (currentAmount + amount > limit)
        {
            // Add only up to the limit
            float actualAdd = Math.Max(0, limit - currentAmount);
            _resourceAmounts[type] = limit;
            
            EmitSignal(SignalName.ResourceAmountChanged, type, limit);
            EmitSignal(SignalName.StorageLimitReached, type);
            
            return actualAdd > 0;
        }
        
        _resourceAmounts[type] = currentAmount + amount;
        EmitSignal(SignalName.ResourceAmountChanged, type, _resourceAmounts[type]);
        CheckUnlockConditions();
        
        return true;
    }

    /// <summary>
    /// Removes resources if available
    /// </summary>
    public bool RemoveResource(ResourceType type, float amount)
    {
        float currentAmount = GetResourceAmount(type);
        if (currentAmount < amount)
            return false;

        _resourceAmounts[type] = currentAmount - amount;
        EmitSignal(SignalName.ResourceAmountChanged, type, _resourceAmounts[type]);
        return true;
    }

    /// <summary>
    /// Gets the storage limit for a resource
    /// </summary>
    public float GetStorageLimit(ResourceType type)
    {
        return _resourceConfigs.TryGetValue(type, out var config) 
            ? config.StorageLimit 
            : float.MaxValue;
    }

    /// <summary>
    /// Sets the storage limit for a resource
    /// </summary>
    public void SetStorageLimit(ResourceType type, float limit)
    {
        if (_resourceConfigs.TryGetValue(type, out var config))
        {
            config.StorageLimit = limit;
            // If current amount exceeds new limit, reduce it
            if (GetResourceAmount(type) > limit)
            {
                _resourceAmounts[type] = limit;
                EmitSignal(SignalName.ResourceAmountChanged, type, limit);
                EmitSignal(SignalName.StorageLimitReached, type);
            }
        }
    }
}

/// <summary>
/// Defines methods for collecting resources
/// </summary>
public interface IResourceCollector
{
    Dictionary<ResourceType, float> Collect();
    bool CanCollect(ResourceType resourceType);
}

/// <summary>
/// Base collector implementation
/// </summary>
public abstract class BaseCollector : IResourceCollector
{
    protected readonly HashSet<ResourceType> CollectableResources = new();
    protected readonly Dictionary<ResourceType, float> CollectionRates = new();

    public virtual Dictionary<ResourceType, float> Collect()
    {
        return CollectionRates.ToDictionary(kvp => kvp.Key, kvp => kvp.Value);
    }

    public bool CanCollect(ResourceType resourceType)
    {
        return CollectableResources.Contains(resourceType);
    }
}

/// <summary>
/// Manages all resources in the game
/// </summary>
public partial class ResourceManager : Node
{
    // Signals
    [Signal]
    public delegate void ResourceAmountChangedEventHandler(ResourceType type, float newAmount);
    
    [Signal]
    public delegate void ResourceUnlockedEventHandler(ResourceType type);
    
    [Signal]
    public delegate void ResourceLockedEventHandler(ResourceType type);

    // Resource state tracking
    private readonly Dictionary<ResourceType, float> _resourceAmounts = new();
    private readonly Dictionary<ResourceType, bool> _unlockedResources = new();
    private readonly Dictionary<ResourceType, ResourceConfig> _resourceConfigs = new();
    private readonly Dictionary<CollectorType, IResourceCollector> _collectors = new();

    public override void _Ready()
    {
        InitializeResources();
        InitializeCollectors();
    }

    /// <summary>
    /// Initializes resource configurations and states
    /// </summary>
    private void InitializeResources()
    {
        // Example configuration for resources
        foreach (ResourceType type in Enum.GetValues(typeof(ResourceType)))
        {
            _resourceAmounts[type] = 0;
            _unlockedResources[type] = type == ResourceType.Resource_1; // Only Resource_1 starts unlocked
            
            var config = new ResourceConfig
            {
                Name = type.ToString(),
                Description = $"Description for {type}",
                IconPath = $"res://assets/icons/{type.ToString().ToLower()}.png"
            };

            // Configure unlock requirements
            if (type != ResourceType.Resource_1)
            {
                ConfigureUnlockRequirements(type, config);
            }

            _resourceConfigs[type] = config;
        }
    }

    /// <summary>
    /// Configures unlock requirements for resources
    /// </summary>
    private void ConfigureUnlockRequirements(ResourceType type, ResourceConfig config)
    {
        // Example of setting up progressive unlock requirements
        int resourceIndex = (int)type;
        for (int i = 0; i < resourceIndex; i++)
        {
            var requiredType = (ResourceType)i;
            float requirement = 1000 * (float)Math.Pow(2, i); // Exponential scaling
            config.UnlockRequirements[requiredType] = requirement;
        }
    }

    /// <summary>
    /// Initializes resource collectors
    /// </summary>
    private void InitializeCollectors()
    {
        // Register collectors
        _collectors[CollectorType.BasicCollector] = new BasicCollector();
        _collectors[CollectorType.AdvancedCollector] = new AdvancedCollector();
        _collectors[CollectorType.MultiCollector] = new MultiCollector();
    }

    /// <summary>
    /// Attempts to collect resources using a specific collector
    /// </summary>
    public bool CollectResources(CollectorType collectorType, ResourceType targetResource)
    {
        if (!_collectors.TryGetValue(collectorType, out var collector))
            return false;

        if (!collector.CanCollect(targetResource) || !_unlockedResources[targetResource])
            return false;

        var collected = collector.Collect();
        foreach (var (resourceType, amount) in collected)
        {
            AddResource(resourceType, amount);
        }

        return true;
    }

    /// <summary>
    /// Adds an amount of a resource and checks unlock conditions
    /// </summary>
    public void AddResource(ResourceType type, float amount)
    {
        _resourceAmounts[type] += amount;
        EmitSignal(SignalName.ResourceAmountChanged, type, _resourceAmounts[type]);
        CheckUnlockConditions();
    }

    /// <summary>
    /// Checks if any locked resources can be unlocked
    /// </summary>
    private void CheckUnlockConditions()
    {
        foreach (var type in Enum.GetValues(typeof(ResourceType)).Cast<ResourceType>())
        {
            if (_unlockedResources[type])
                continue;

            if (CanUnlock(type))
            {
                UnlockResource(type);
            }
        }
    }

    /// <summary>
    /// Checks if a resource can be unlocked
    /// </summary>
    public bool CanUnlock(ResourceType type)
    {
        if (_unlockedResources[type])
            return false;

        var requirements = _resourceConfigs[type].UnlockRequirements;
        return requirements.All(req => _resourceAmounts[req.Key] >= req.Value);
    }

    /// <summary>
    /// Unlocks a resource
    /// </summary>
    private void UnlockResource(ResourceType type)
    {
        _unlockedResources[type] = true;
        EmitSignal(SignalName.ResourceUnlocked, type);
    }

    /// <summary>
    /// Gets the current amount of a resource
    /// </summary>
    public float GetResourceAmount(ResourceType type)
    {
        return _resourceAmounts.GetValueOrDefault(type, 0);
    }

    /// <summary>
    /// Gets the configuration for a resource
    /// </summary>
    public ResourceConfig GetResourceConfig(ResourceType type)
    {
        return _resourceConfigs.GetValueOrDefault(type);
    }

    /// <summary>
    /// Checks if a resource is unlocked
    /// </summary>
    public bool IsResourceUnlocked(ResourceType type)
    {
        return _unlockedResources.GetValueOrDefault(type, false);
    }
}
