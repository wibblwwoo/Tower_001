using System;

/// <summary>
/// Utility class for calculating completion times based on different rate patterns
/// </summary>
public static class RateCalculator
{
    /// <summary>
    /// Calculates time to reach a target value with a constant rate
    /// </summary>
    /// <param name="currentValue">Starting value</param>
    /// <param name="targetValue">Value to reach</param>
    /// <param name="ratePerSecond">Units gained per second</param>
    /// <returns>Time in seconds to reach target</returns>
    public static float CalculateLinearCompletionTime(float currentValue, float targetValue, float ratePerSecond)
    {
        if (ratePerSecond <= 0)
            throw new ArgumentException("Rate must be greater than 0", nameof(ratePerSecond));
            
        float remaining = targetValue - currentValue;
        return remaining / ratePerSecond;
    }
    
    /// <summary>
    /// Calculates time to reach a target with an accelerating rate
    /// </summary>
    /// <param name="currentValue">Starting value</param>
    /// <param name="targetValue">Value to reach</param>
    /// <param name="initialRate">Starting rate per interval</param>
    /// <param name="rateIncrease">How much the rate increases per interval</param>
    /// <param name="intervalInSeconds">Time between rate increases</param>
    /// <returns>Time in seconds to reach target</returns>
    public static float CalculateAcceleratingCompletionTime(
        float currentValue,
        float targetValue,
        float initialRate,
        float rateIncrease,
        float intervalInSeconds)
    {
        if (initialRate <= 0)
            throw new ArgumentException("Initial rate must be greater than 0", nameof(initialRate));
            
        if (intervalInSeconds <= 0)
            throw new ArgumentException("Interval must be greater than 0", nameof(intervalInSeconds));
            
        float remainingValue = targetValue - currentValue;
        float intervals = 0;
        float currentRate = initialRate;
        float accumulated = 0;
        
        while (accumulated < remainingValue)
        {
            accumulated += currentRate;
            currentRate += rateIncrease;
            intervals++;
        }
        
        return intervals * intervalInSeconds;
    }
    
    /// <summary>
    /// Calculates time to reach a target with a compound growth rate
    /// </summary>
    /// <param name="currentValue">Starting value</param>
    /// <param name="targetValue">Value to reach</param>
    /// <param name="initialRate">Starting rate per interval</param>
    /// <param name="growthMultiplier">Rate multiplier per interval (e.g., 1.1 for 10% growth)</param>
    /// <param name="intervalInSeconds">Time between rate increases</param>
    /// <returns>Time in seconds to reach target</returns>
    public static float CalculateCompoundingCompletionTime(
        float currentValue,
        float targetValue,
        float initialRate,
        float growthMultiplier,
        float intervalInSeconds)
    {
        if (initialRate <= 0)
            throw new ArgumentException("Initial rate must be greater than 0", nameof(initialRate));
            
        if (growthMultiplier <= 1)
            throw new ArgumentException("Growth multiplier must be greater than 1", nameof(growthMultiplier));
            
        float remainingValue = targetValue - currentValue;
        float intervals = 0;
        float currentRate = initialRate;
        float accumulated = 0;
        
        while (accumulated < remainingValue)
        {
            accumulated += currentRate;
            currentRate *= growthMultiplier;
            intervals++;
        }
        
        return intervals * intervalInSeconds;
    }
}
