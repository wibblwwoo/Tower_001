# InventorySystem/Controls/InventorySlot.cs
public partial class InventorySlot : Panel
{
    public int SlotIndex { get; private set; }
    public bool IsEmpty => Item == null;
    public InventoryItem Item { get; private set; }
    
    [Signal]
    public delegate void ItemAddedEventHandler(InventoryItem item);
    [Signal]
    public delegate void ItemRemovedEventHandler(InventoryItem item);
    
    public InventorySlot(int index)
    {
        SlotIndex = index;
        CustomMinimumSize = new Vector2(64, 64);
        
        // Add visual style
        AddThemeStylebox("panel", new StyleBoxFlat
        {
            BgColor = new Color(0.2f, 0.2f, 0.2f),
            BorderWidthLeft = 1,
            BorderWidthTop = 1,
            BorderWidthRight = 1,
            BorderWidthBottom = 1,
            BorderColor = new Color(0.3f, 0.3f, 0.3f)
        });
        
        // Setup drag and drop
        SetupDragDrop();
    }
    
    private void SetupDragDrop()
    {
        GuiInput += (InputEvent @event) =>
        {
            if (@event is InputEventMouseButton mouseEvent)
            {
                if (mouseEvent.ButtonIndex == MouseButton.Left)
                {
                    if (mouseEvent.Pressed)
                    {
                        StartDrag();
                    }
                    else
                    {
                        EndDrag();
                    }
                }
            }
        };
    }
    
    private void StartDrag()
    {
        if (!IsEmpty)
        {
            // Start drag operation
            var dragData = new SceneTreeDragData();
            dragData.Data = new Dictionary<string, object>
            {
                { "type", "inventory_item" },
                { "item", Item },
                { "source_slot", this }
            };
            
            SetDragPreview(CreateDragPreview());
        }
    }
    
    private Control CreateDragPreview()
    {
        var preview = new TextureRect
        {
            Texture = Item.Icon,
            CustomMinimumSize = new Vector2(48, 48)
        };
        
        return preview;
    }
    
    private void EndDrag()
    {
        // Handle in the grid
    }
    
    public void SetItem(InventoryItem item)
    {
        Item = item;
        if (item != null)
        {
            EmitSignal(SignalName.ItemAdded, item);
            UpdateVisual();
        }
    }
    
    public InventoryItem RemoveItem()
    {
        var item = Item;
        Item = null;
        EmitSignal(SignalName.ItemRemoved, item);
        UpdateVisual();
        return item;
    }
    
    private void UpdateVisual()
    {
        // Clear existing visuals
        foreach (var child in GetChildren())
        {
            RemoveChild(child);
            child.QueueFree();
        }
        
        if (Item != null)
        {
            var textureRect = new TextureRect
            {
                Texture = Item.Icon,
                CustomMinimumSize = new Vector2(48, 48),
                SizeFlagsHorizontal = SizeFlags.Expand | SizeFlags.Fill,
                SizeFlagsVertical = SizeFlags.Expand | SizeFlags.Fill,
                StretchMode = TextureRect.StretchModeEnum.KeepAspectCentered
            };
            
            AddChild(textureRect);
        }
    }
}

# InventorySystem/Controls/InventoryGrid.cs
[UIState(UIStates.GAME_MENU, "Inventory", UILayerType.Window)]
public partial class InventoryGrid : Panel, IUIState
{
    public string StateName => UIStates.GAME_MENU;
    public UILayerType PreferredLayer => UILayerType.Window;
    public bool IsVisible { get; set; }
    
    private GridContainer _grid;
    private List<InventorySlot> _slots = new List<InventorySlot>();
    
    [Export]
    public int Rows { get; set; } = 5;
    
    [Export]
    public int Columns { get; set; } = 6;
    
    [Signal]
    public delegate void ItemMovedEventHandler(int fromSlot, int toSlot);
    
    public override void _Ready()
    {
        // Set up the grid container
        _grid = new GridContainer
        {
            Columns = Columns
        };
        
        AddChild(_grid);
        
        // Create slots
        InitializeGrid();
        
        // Add some padding and style
        var stylebox = new StyleBoxFlat
        {
            BgColor = new Color(0.15f, 0.15f, 0.15f),
            ContentMarginLeft = 10,
            ContentMarginTop = 10,
            ContentMarginRight = 10,
            ContentMarginBottom = 10
        };
        
        AddThemeStylebox("panel", stylebox);
        
        // Setup drag and drop for the grid
        SetupGridDragDrop();
    }
    
    private void InitializeGrid()
    {
        for (int i = 0; i < Rows * Columns; i++)
        {
            var slot = new InventorySlot(i);
            _slots.Add(slot);
            _grid.AddChild(slot);
        }
    }
    
    private void SetupGridDragDrop()
    {
        foreach (var slot in _slots)
        {
            slot.GuiInput += (@event) =>
            {
                if (@event is InputEventMouseButton mouseEvent && 
                    mouseEvent.ButtonIndex == MouseButton.Left && 
                    !mouseEvent.Pressed)
                {
                    HandleDrop(slot);
                }
            };
        }
    }
    
    private void HandleDrop(InventorySlot targetSlot)
    {
        var dragData = GetViewport().GuiGetDragData();
        if (dragData != null && 
            dragData.Data is Dictionary<string, object> data && 
            data.ContainsKey("type") && 
            data["type"].ToString() == "inventory_item")
        {
            var sourceSlot = data["source_slot"] as InventorySlot;
            if (sourceSlot != null && sourceSlot != targetSlot)
            {
                SwapItems(sourceSlot, targetSlot);
            }
        }
    }
    
    private void SwapItems(InventorySlot sourceSlot, InventorySlot targetSlot)
    {
        var sourceItem = sourceSlot.RemoveItem();
        var targetItem = targetSlot.RemoveItem();
        
        if (targetItem != null)
        {
            sourceSlot.SetItem(targetItem);
        }
        
        if (sourceItem != null)
        {
            targetSlot.SetItem(sourceItem);
        }
        
        EmitSignal(SignalName.ItemMoved, sourceSlot.SlotIndex, targetSlot.SlotIndex);
    }
    
    public void AddItem(InventoryItem item, int slotIndex)
    {
        if (slotIndex >= 0 && slotIndex < _slots.Count && _slots[slotIndex].IsEmpty)
        {
            _slots[slotIndex].SetItem(item);
        }
    }
    
    public InventoryItem RemoveItem(int slotIndex)
    {
        if (slotIndex >= 0 && slotIndex < _slots.Count && !_slots[slotIndex].IsEmpty)
        {
            return _slots[slotIndex].RemoveItem();
        }
        return null;
    }
    
    public void Show()
    {
        Visible = true;
        IsVisible = true;
    }
    
    public void Hide()
    {
        Visible = false;
        IsVisible = false;
    }
}

            Rows = 5,
            Columns = 6
        };
        
        AddChild(_inventoryGrid);
        
        // Example: Add some test items

            ItemName = "Steel Sword",

            IsStackable = false
        };
        
        var potionItem = new InventoryItem
        {
            ItemName = "Health Potion",

            IsStackable = true,
            MaxStackSize = 10
        };
        
        _inventoryGrid.AddItem(swordItem, 0);
        _inventoryGrid.AddItem(potionItem, 1);
        
        // Connect signals
        _inventoryGrid.ItemMoved += (fromSlot, toSlot) =>
        {
            GD.Print($"Item moved from slot {fromSlot} to slot {toSlot}");
        };
    }
